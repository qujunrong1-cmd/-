# -*- coding: utf-8 -*-
"""同步游戏版本数据：从 valorant-api.com 拉取当前版本、全部特工、全部武器（中文），
与本地 data/terms.json 比对，输出差异并把本地名词表更新到最新。
用法：python sync_meta.py [--write]
不带 --write 只做对比（dry-run）。
"""
import json, os, sys, datetime

ROOT = r"E:\无畏契约分析"
DATA = os.path.join(ROOT, "data")
API = "https://valorant-api.com/v1"
UA = {"User-Agent": "Mozilla/5.0 (compatible; ValorantMetaSync/1.0)"}
WRITE = "--write" in sys.argv

CAT_MAP = {"佩枪": "佩枪", "冲锋枪": "冲锋枪", "霰弹枪": "霰弹枪", "狙击枪": "狙击枪",
           "突击步枪": "步枪", "步枪": "步枪", "重武器": "机枪", "机枪": "机枪"}


def get(path):
    """本机需要走本地代理，urllib 的 SSL 握手会超时，改用 curl 更稳。"""
    import subprocess
    out = subprocess.run(["curl", "-s", "--max-time", "40", "--retry", "2",
                          "-H", UA["User-Agent"], API + path],
                         capture_output=True, timeout=120)
    if out.returncode != 0 or not out.stdout:
        raise RuntimeError(f"拉取失败: {path} rc={out.returncode}")
    return json.loads(out.stdout.decode("utf-8"))


def main():
    print("① 拉取版本信息…")
    ver = get("/version")["data"]
    print(f"   客户端版本 {ver['version']}   分支 {ver['branch']}   构建日期 {ver['buildDate'][:10]}")

    print("② 拉取特工列表…")
    agents = [a for a in get("/agents?language=zh-CN&isPlayableCharacter=true")["data"]
              if a.get("isPlayableCharacter")]
    api_agents = {}
    for a in agents:
        role = (a.get("role") or {}).get("displayName", "").strip() or "未知"
        api_agents[a["displayName"]] = {"en": a.get("displayName") and a.get("assetName", ""),
                                        "role": role}
    print(f"   {len(api_agents)} 名")

    print("③ 拉取武器列表…")
    weapons = get("/weapons?language=zh-CN")["data"]
    api_weapons = {}
    for w in weapons:
        shop = w.get("shopData") or {}
        cat = CAT_MAP.get(shop.get("categoryText", ""), shop.get("categoryText", "—"))
        if w["displayName"] == "近战武器":
            continue
        api_weapons[w["displayName"]] = {"cat": cat, "price": shop.get("cost", None)}
    print(f"   {len(api_weapons)} 把")

    # ---- 与本地比对 ----
    t = json.load(open(os.path.join(DATA, "terms.json"), encoding="utf-8"))
    local_agents = {a["cn"]: a for a in t["agents"]}
    local_weapons = {w["cn"]: w for w in t["weapons"]}

    diffs = []
    new_agents = [n for n in api_agents if n not in local_agents]
    gone_agents = [n for n in local_agents if n not in api_agents]
    role_fix = [(n, local_agents[n]["role"], api_agents[n]["role"]) for n in api_agents
                if n in local_agents and local_agents[n]["role"] != api_agents[n]["role"]]

    new_weapons = [n for n in api_weapons if n not in local_weapons]
    gone_weapons = [n for n in local_weapons if n not in api_weapons]
    price_fix = []
    for n, w in api_weapons.items():
        if n in local_weapons and w["price"] is not None:
            old = str(local_weapons[n]["price"])
            if old != str(w["price"]) and old != "免费":
                price_fix.append((n, old, w["price"]))

    print("\n④ 差异报告")
    print(f"   新特工 {len(new_agents)}: {new_agents or '无'}")
    print(f"   已下架/改名 {len(gone_agents)}: {gone_agents or '无'}")
    print(f"   定位变动 {len(role_fix)}: {role_fix or '无'}")
    print(f"   新武器 {len(new_weapons)}: {new_weapons or '无'}")
    print(f"   武器缺失 {len(gone_weapons)}: {gone_weapons or '无'}")
    print(f"   价格更新 {len(price_fix)}: {price_fix or '无'}")

    # ---- 更新 terms.json ----
    if WRITE:
        for n in new_agents:
            t["agents"].append({"no": len(t["agents"]) + 1, "cn": n,
                                "en": api_agents[n]["en"] or "—",
                                "role": api_agents[n]["role"], "origin": "—"})
        for n, _, role in role_fix:
            for a in t["agents"]:
                if a["cn"] == n:
                    a["role"] = role
        for n, w in api_weapons.items():
            if n in local_weapons:
                local_weapons[n]["price"] = str(w["price"]) if w["price"] is not None else "—"
                local_weapons[n]["cat"] = w["cat"]
            else:
                t["weapons"].append({"cn": n, "en": "—", "cat": w["cat"],
                                     "price": str(w["price"]) if w["price"] is not None else "—",
                                     "note": "自动同步新增"})
        t["gameVersion"] = ver["branch"].replace("release-", "")
        json.dump(t, open(os.path.join(DATA, "terms.json"), "w", encoding="utf-8"),
                  ensure_ascii=False, indent=2)
        print("\n   terms.json 已更新")
    else:
        print("\n   （dry-run，未写入。加 --write 生效）")

    # ---- 写 meta.json ----
    meta = {
        "gameVersion": ver["branch"].replace("release-", ""),
        "clientVersion": ver["version"],
        "buildDate": ver["buildDate"][:10],
        "syncedAt": datetime.datetime.now().strftime("%Y-%m-%d %H:%M"),
        "counts": {"agents": len(api_agents), "weapons": len(api_weapons)},
        "roleSplit": {},
        "syncLog": {
            "newAgents": new_agents, "goneAgents": gone_agents,
            "roleChanged": role_fix, "newWeapons": new_weapons,
            "priceChanged": price_fix,
        },
    }
    for n, a in api_agents.items():
        meta["roleSplit"][a["role"]] = meta["roleSplit"].get(a["role"], 0) + 1
    json.dump(meta, open(os.path.join(DATA, "meta.json"), "w", encoding="utf-8"),
              ensure_ascii=False, indent=2)
    print(f"   meta.json 已写入（{meta['gameVersion']} / 特工 {len(api_agents)} / 武器 {len(api_weapons)}）")
    return meta


if __name__ == "__&#8203;main__":
    main()
